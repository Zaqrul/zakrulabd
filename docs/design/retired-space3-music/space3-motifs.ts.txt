/* The themes of Space 1 and Space 2 that Space 3 quotes. Metal Slug brings its main theme back later in the series in a
   bigger arrangement ("Final Attack"), and Space 3 does the same with its own two predecessors: the chiptune of Space 1 is
   played on the square-wave "chip" voice, the orchestra of Space 2 on brass and strings, so a listener hears the two earlier
   games inside the third. Every line is copied from the score it comes from; tests/space3-motifs.test.mjs compares each one
   with the original, so a change in Space 1 or 2 shows up here. (These are the owner's own themes, so quoting them is
   deliberate; the rest of Space 3 copies nothing.)

   They are written out here, not imported, because the music engine imports Space 3's tunes. */

export type Quoted = { line: string; chord: string };
const q = (line: string, chord: string): Quoted => ({ line, chord });

/* Space 1: Starfall Overture (the title, C major). */
export const S1_TITLE: Quoted[] = [
  q("E5 - - - G5 - C6 - - - B5 - C6 - D6 -", "C"),
  q("B5 - - - G5 - - - D5 - - - G5 - - -", "G"),
  q("C6 - - - A5 - E5 - - - A5 - B5 - C6 -", "Am"),
  q("A5 - - - - - - - F5 - G5 - A5 - - -", "F"),
];

/* Space 1: Battle: First Sortie (the first stage, E minor). */
export const S1_STAGE: Quoted[] = [

  q("G5 - - - - - E5 - C5 - - - E5 - G5 -", "C"),
  q("A5 - - - F#5 - D5 - - - E5 - F#5 - A5 -", "D"),
  q("B5 - - - - - - - A5 - F#5 - D#5 - - -", "B7"),
];
